import duckdb

cur='/work/candidate/curated.parquet'
raw='/work/envelopes.csv'

SELECT
  c.record_id,
  c.source_object,
  c.source_row,
  c.source_sha256,
  r.parse_ok
FROM read_parquet('{cur}') c
JOIN read_csv_auto('{raw}') r
  ON c.source_object = r.source_object
 AND c.source_row = r.source_row
 AND c.source_sha256 = r.source_sha256
ORDER BY c.record_id
LIMIT 1
''').fetchone()

print(r)
